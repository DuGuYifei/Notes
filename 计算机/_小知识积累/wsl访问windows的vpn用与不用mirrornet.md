# wsl访问windows的vpn用与不用mirrornet

## mirrornet

开启

```powershell
notepad $env:USERPROFILE\.wslconfig
```

```
[wsl2]
networkingMode=mirrored
dnsTunneling=true
autoProxy=true
firewall=true
```

## 不用mirrornet

```bash
powershell.exe -ExecutionPolicy Bypass -File .\setup-wsl-proxy.ps1
```

```bash
source ~/.config/wsl-windows-proxy.sh
env | grep -iE '^(http|https|all|no)_proxy='
curl -I --connect-timeout 5 --max-time 15 https://www.google.com
```

```powershell
[CmdletBinding()]
param(
    [string]$Distro = ""
)

$ErrorActionPreference = "Stop"

function Test-IsAdministrator {
    $identity = [Security.Principal.WindowsIdentity]::GetCurrent()
    $principal = New-Object Security.Principal.WindowsPrincipal($identity)
    return $principal.IsInRole([Security.Principal.WindowsBuiltInRole]::Administrator)
}

function Read-Port {
    param(
        [string]$Prompt,
        [int]$Default
    )

    $value = Read-Host "$Prompt [$Default]"
    if ([string]::IsNullOrWhiteSpace($value)) {
        return $Default
    }

    $port = 0
    if (-not [int]::TryParse($value, [ref]$port) -or $port -lt 1 -or $port -gt 65535) {
        throw "Invalid port: $value"
    }

    return $port
}

if (-not (Test-IsAdministrator)) {
    throw "Run this script from an Administrator PowerShell."
}

$HttpPort  = Read-Port "Windows HTTP(S) proxy port" 11304
$SocksPort = Read-Port "Windows SOCKS5 proxy port" 11305

# Find Fedora WSL automatically unless -Distro is provided.
$distros = @(
    (& wsl.exe --list --quiet) |
        ForEach-Object { ($_ -replace "`0", "").Trim() } |
        Where-Object { $_ }
)

if (-not $Distro) {
    $fedora = @($distros | Where-Object { $_ -match '(?i)fedora' })

    if ($fedora.Count -eq 1) {
        $Distro = $fedora[0]
    }
    elseif ($fedora.Count -gt 1) {
        throw "Multiple Fedora WSL distros found: $($fedora -join ', '). Re-run with -Distro <name>."
    }
    else {
        throw "No Fedora WSL distro found. Installed distros: $($distros -join ', ')"
    }
}

if ($distros -notcontains $Distro) {
    throw "WSL distro not found: $Distro"
}

# Read the NAT default route and extract the Windows-side gateway.
$routeOutput = (& wsl.exe -d $Distro -- ip route show default) -join "`n"
if ($routeOutput -notmatch '\bdefault\s+via\s+([0-9]+(?:\.[0-9]+){3})\b') {
    throw "Could not determine the WSL NAT gateway. Current default route: $routeOutput"
}
$gateway = $Matches[1]

Write-Host ""
Write-Host "WSL distro      : $Distro"
Write-Host "Windows gateway : $gateway"

# netsh portproxy depends on the IP Helper service.
Set-Service -Name iphlpsvc -StartupType Automatic
Start-Service -Name iphlpsvc

# Bridge:
#   WSL -> WindowsGateway:11304 -> Windows localhost:11304
#   WSL -> WindowsGateway:11305 -> Windows localhost:11305
foreach ($port in @($HttpPort, $SocksPort)) {
    & netsh.exe interface portproxy delete v4tov4 `
        listenaddress=$gateway `
        listenport=$port 2>$null | Out-Null

    & netsh.exe interface portproxy add v4tov4 `
        listenaddress=$gateway `
        listenport=$port `
        connectaddress=127.0.0.1 `
        connectport=$port | Out-Null

    if ($LASTEXITCODE -ne 0) {
        throw "Failed to create portproxy for port $port."
    }
}

# Allow only the two bridge ports on the Windows-side WSL gateway.
$ruleName = "WSL Windows Proxy Bridge"

Get-NetFirewallRule -DisplayName $ruleName -ErrorAction SilentlyContinue |
    Remove-NetFirewallRule -ErrorAction SilentlyContinue

New-NetFirewallRule `
    -DisplayName $ruleName `
    -Direction Inbound `
    -Action Allow `
    -Protocol TCP `
    -LocalAddress $gateway `
    -LocalPort @("$HttpPort", "$SocksPort") `
    -Profile Any | Out-Null

# This helper runs ONCE when a new bash/zsh shell starts.
# If Windows system proxy is enabled and the local proxy ports are listening,
# it exports proxy environment variables for that shell.
# Otherwise, the shell uses normal direct networking.
$helper = @'
# Generated by setup-wsl-proxy-fixed-full.ps1

_WSL_HTTP_PORT=__HTTP_PORT__
_WSL_SOCKS_PORT=__SOCKS_PORT__
_REG_EXE="/mnt/c/Windows/System32/reg.exe"
_NETSTAT_EXE="/mnt/c/Windows/System32/netstat.exe"

_windows_proxy_enabled() {
    "$_REG_EXE" query \
        'HKCU\Software\Microsoft\Windows\CurrentVersion\Internet Settings' \
        /v ProxyEnable 2>/dev/null |
        tr -d '\r' |
        grep -Ei 'ProxyEnable[[:space:]]+REG_DWORD[[:space:]]+0x1' >/dev/null
}

_windows_port_listening() {
    local port="$1"
    "$_NETSTAT_EXE" -ano -p TCP 2>/dev/null |
        tr -d '\r' |
        grep -Ei "[:.]${port}[[:space:]].*LISTENING" >/dev/null
}

# Start clean for this shell.
unset HTTP_PROXY HTTPS_PROXY http_proxy https_proxy
unset ALL_PROXY all_proxy

if _windows_proxy_enabled; then
    _gw="$(ip route show default 2>/dev/null | awk 'NR==1 {print $3}')"

    if [ -n "$_gw" ]; then
        if _windows_port_listening "$_WSL_HTTP_PORT"; then
            export HTTP_PROXY="http://${_gw}:${_WSL_HTTP_PORT}"
            export HTTPS_PROXY="$HTTP_PROXY"
            export http_proxy="$HTTP_PROXY"
            export https_proxy="$HTTPS_PROXY"
        fi

        if _windows_port_listening "$_WSL_SOCKS_PORT"; then
            export ALL_PROXY="socks5h://${_gw}:${_WSL_SOCKS_PORT}"
            export all_proxy="$ALL_PROXY"
        fi

        case ",${NO_PROXY:-}," in
            *,localhost,*) ;;
            *) export NO_PROXY="localhost,127.0.0.1,::1${NO_PROXY:+,$NO_PROXY}" ;;
        esac
        export no_proxy="$NO_PROXY"
    fi
fi

unset _gw
'@

$helper = $helper.Replace("__HTTP_PORT__", [string]$HttpPort)
$helper = $helper.Replace("__SOCKS_PORT__", [string]$SocksPort)

$helperBytes = [System.Text.Encoding]::UTF8.GetBytes($helper)
$helperB64 = [Convert]::ToBase64String($helperBytes)

$installer = @'
set -eu

mkdir -p "$HOME/.config"
printf '%s' '__HELPER_B64__' | base64 -d > "$HOME/.config/wsl-windows-proxy.sh"
chmod 600 "$HOME/.config/wsl-windows-proxy.sh"

line='[ -f "$HOME/.config/wsl-windows-proxy.sh" ] && . "$HOME/.config/wsl-windows-proxy.sh"'

touch "$HOME/.bashrc"
grep -Fqx "$line" "$HOME/.bashrc" || printf '\n%s\n' "$line" >> "$HOME/.bashrc"

touch "$HOME/.zshrc"
grep -Fqx "$line" "$HOME/.zshrc" || printf '\n%s\n' "$line" >> "$HOME/.zshrc"
'@

$installer = $installer.Replace("__HELPER_B64__", $helperB64)
$installerBytes = [System.Text.Encoding]::UTF8.GetBytes($installer)
$installerB64 = [Convert]::ToBase64String($installerBytes)

& wsl.exe -d $Distro -- bash -lc "printf '%s' '$installerB64' | base64 -d | bash"

if ($LASTEXITCODE -ne 0) {
    throw "Failed to install the WSL shell helper."
}

Write-Host ""
Write-Host "Done."
Write-Host "HTTP(S): $gateway`:$HttpPort -> 127.0.0.1:$HttpPort"
Write-Host "SOCKS5 : $gateway`:$SocksPort -> 127.0.0.1:$SocksPort"
Write-Host ""
Write-Host "Open a NEW WSL terminal."
Write-Host "At shell startup:"
Write-Host "  Windows proxy ON  -> HTTP_PROXY / HTTPS_PROXY / ALL_PROXY are exported."
Write-Host "  Windows proxy OFF -> normal direct WSL networking."
```
